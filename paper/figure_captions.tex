\begin{figure}[t]
    \centering
    \includegraphics[width=\linewidth]{experiments/08_arbitrage_frontier/results/kdd_rational_boundary.png}
    \caption{\textbf{The Arbitrage Frontier (Learned Policy).} 
    Visualization of the \texttt{BanditRouter}'s decision manifold between \texttt{gpt-oss-120b} and \texttt{gemini-3-pro}. 
    The Y-axis represents the market cost premium ($\Delta C \approx \$6$/1M tokens), exhibiting minor variance due to prompt-specific input/output ratios. 
    The X-axis shows the predicted quality lift ($\Delta Q$). 
    The dashed line represents the theoretical indifference curve (Slope $1/\lambda = 50.0$). 
    The router successfully identifies ``Arbitrage Opportunities'' (Red)—prompts where the intrinsic utility gain exceeds the market price premium—while relegating commodity tasks (Blue) to the efficient baseline.
    }
    \label{fig:rational_boundary}
\end{figure}

\begin{figure}[t]
    \centering
    \includegraphics[width=\linewidth]{experiments/11_routellm_comparison/results/pareto_frontier.png}
    \caption{\textbf{Cost-Quality Pareto Frontier: BanditGPT vs. RouteLLM.}
    Direct comparison of routing performance between BanditGPT (red stars) and RouteLLM (blue circles) on a two-model task (\texttt{gpt-oss-120b} vs. \texttt{gemini-3-pro-preview}).
    X-axis: Normalized cost relative to flagship-only deployment (0\% = weak model only, 100\% = strong model only).
    Y-axis: Recovered quality relative to flagship performance (100\% = perfect routing).
    BanditGPT achieves Pareto dominance across the 25-75\% cost range, demonstrating +3.1\% quality advantage at the 50\% cost operating point (98.2\% vs. 95.1\%, $p < 0.05$).
    The gray dashed line shows random baseline; green and gold squares mark single-model performance anchors.
    Error bars: $\pm$1 std across 5 trials.
    This proves BanditGPT delivers ``more quality for less money'' than the industry-standard RouteLLM baseline.
    }
    \label{fig:pareto_frontier}
\end{figure}
