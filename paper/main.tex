\begin{figure}[t]

\caption{Cumulative regret: warmup priors vs.\ cold start across three budget regimes (K=3 stationary, 20 seeds, 95\% bootstrap CI, held-out test split).
  Left: unconstrained---warmup reduces regret by 11\% ($p < 10^{-5}$), concentrated in the first ${\sim}200$ steps.
  Centre: tight budget---the advantage persists ($\Delta{=}-7.3$, $p{=}0.006$) because uninformed UCB exploration wastes scarce budget on expensive arms.
  Right: moderate budget---the gap vanishes ($p{=}0.68$) as the budget constraint collapses the feasible arm set, making the prior's knowledge irrelevant.}

\begin{figure}[t]
\centering
\includegraphics[width=0.85\linewidth]{../experiments/appendix/warmup_ablation/results/warmup_uncertainty.pdf}
\caption{Uncertainty evolution for warmup vs.\ cold-start conditions (K=3 stationary, 20 seeds, 95\% bootstrap CI).
  $\mathrm{tr}(A_a^{-1})$ measures total per-arm uncertainty; higher values yield larger UCB exploration bonuses.
  Solid lines: warmup-initialised arms start at $\mathrm{tr}(A_a^{-1}) \approx 0.72$, reflecting the $n_\mathit{eff}{=}1{,}000$ pseudo-observations encoded in the prior.
  Dashed lines: cold-started arms begin at $\mathrm{tr}(A_a^{-1}) = d/\lambda = 26$---a ${\sim}36{\times}$ uncertainty gap that forces hundreds of exploratory selections before the router can make informed decisions.
  With geometric forgetting ($\gamma{=}0.995$), arms selected infrequently see rising uncertainty as the prior decays.
  Frequently selected arms converge by step~${\sim}400$, consistent with the regret convergence in Figure~\ref{fig:warmup_ablation}.}
\label{fig:warmup_uncertainty}
\end{figure}
