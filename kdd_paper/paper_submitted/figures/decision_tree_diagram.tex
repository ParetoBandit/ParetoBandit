% Decision Tree Diagram - Routing Logic Flow
\documentclass[tikz,border=10pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{shapes,arrows,positioning,calc,fit,backgrounds,shapes.geometric}

\begin{document}

\tikzstyle{start} = [ellipse, draw, fill=green!30, text width=5em, text centered, minimum height=2.5em]
\tikzstyle{process} = [rectangle, draw, fill=blue!20, text width=7em, text centered, rounded corners, minimum height=2.5em]
\tikzstyle{decision} = [diamond, draw, fill=yellow!20, text width=5em, text centered, minimum height=3em, aspect=2]
\tikzstyle{model} = [rectangle, draw, fill=orange!20, text width=6em, text centered, rounded corners, minimum height=2em]
\tikzstyle{end} = [ellipse, draw, fill=red!20, text width=5em, text centered, minimum height=2.5em]
\tikzstyle{arrow} = [thick,->,>=stealth]

\begin{tikzpicture}[node distance=1.3cm, auto]

% Start
\node [start] (query) {User Query\\$q$};

% Feature Extraction
\node [process, below=of query] (embed) {Extract Features\\$\mathbf{x} = \phi(q)$\\{\small 384-dim}};

% Mode Decision
\node [decision, below=of embed] (mode) {Operating\\Mode?};

% STANDARD MODE PATH (Left)
\node [process, below left=1cm and 2cm of mode, fill=blue!30] (ucb_standard) {Compute UCB\\for all models\\$m \in \mathcal{M}$};

\node [process, below=of ucb_standard] (calc_standard) {$\text{UCB}_m = $\\$\hat{Q}(\mathbf{x},m)$\\$+ \beta \sigma_m(\mathbf{x})$\\$- \lambda_c C_m$};

\node [process, below=of calc_standard] (select_standard) {$m^* = \argmax$\\$\text{UCB}_m$};

\node [model, below=of select_standard] (route_standard) {Route to $m^*$\\{\small Single-shot}};

% HYBRID MODE PATH (Right)
\node [process, below right=1cm and 2cm of mode, fill=purple!30] (ucb_hybrid) {Compute UCB\\+ Uncertainty\\$\sigma_m(\mathbf{x})$};

\node [decision, below=of ucb_hybrid] (confidence) {$\max \text{UCB}_m$\\$> \tau_{verify}$?};

\node [process, below left=1cm and 1cm of confidence] (single) {Route to\\Best Model\\{\small High confidence}};

\node [process, below right=1cm and 1cm of confidence] (cascade) {Cascade:\\Best → Verify\\{\small Low confidence}};

% Model Execution (Converge paths)
\node [process, below=3cm of route_standard, fill=orange!30, text width=8em] (execute) {Execute on\\Selected Model(s)\\Get Response};

% Feedback
\node [process, below=of execute] (feedback) {Compute Reward\\$r = Q - \lambda_c C$\\{\small Update bandit}};

% End
\node [end, below=of feedback] (done) {Return\\Response};

% Arrows - Main Flow
\draw [arrow] (query) -- (embed);
\draw [arrow] (embed) -- (mode);

% Arrows - Standard Mode
\draw [arrow] (mode) -| node[near start, above left] {\small Standard} (ucb_standard);
\draw [arrow] (ucb_standard) -- (calc_standard);
\draw [arrow] (calc_standard) -- (select_standard);
\draw [arrow] (select_standard) -- (route_standard);
\draw [arrow] (route_standard) |- (execute);

% Arrows - Hybrid Mode
\draw [arrow] (mode) -| node[near start, above right] {\small Hybrid} (ucb_hybrid);
\draw [arrow] (ucb_hybrid) -- (confidence);
\draw [arrow] (confidence) -| node[near start, above] {\small Yes} (single);
\draw [arrow] (confidence) -| node[near start, above] {\small No} (cascade);
\draw [arrow] (single) |- (execute);
\draw [arrow] (cascade) |- (execute);

% Arrows - Feedback
\draw [arrow] (execute) -- (feedback);
\draw [arrow] (feedback) -- (done);

% Feedback loop back to bandit
\draw [arrow, dashed, red, thick] (feedback.west) -- ++(-1.5,0) |- node[near start, left, text width=2cm, align=center, font=\scriptsize] {Update\\$\mathbf{A}_m$\\$\mathbf{b}_m$} (embed.west);

% Annotations - Standard Mode
\node[draw, dashed, fit=(ucb_standard) (calc_standard) (select_standard) (route_standard), inner sep=0.3cm, label=above left:\textbf{Standard Mode}, fill=blue!5, opacity=0.5] (standard_box) {};

\node[text width=3cm, align=left, font=\scriptsize] at (-5, -6) {
\textbf{Cost Leader}\\
• Minimize cost\\
• Acceptable quality\\
• O(1) latency
};

% Annotations - Hybrid Mode
\node[draw, dashed, fit=(ucb_hybrid) (confidence) (single) (cascade), inner sep=0.3cm, label=above right:\textbf{Hybrid Mode}, fill=purple!5, opacity=0.5] (hybrid_box) {};

\node[text width=3cm, align=left, font=\scriptsize] at (5, -6) {
\textbf{High Assurance}\\
• Verify uncertainty\\
• Max reliability\\
• Selective cascade
};

% Constraints box
\node[draw, thick, fit=(mode), inner sep=0.6cm, dashed, blue] (constraints) {};
\node[above=0.2cm of constraints, text width=4cm, align=center, font=\scriptsize\bfseries] {
User Constraints:\\
$\lambda_{cost}$, $\lambda_{latency}$, $\tau_{verify}$
};

\end{tikzpicture}

\end{document}
