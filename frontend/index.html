Core Innovation

5D Chebyshev with Net Pareto Dominance

Extends 4D optimization with quality-weighted Net Pareto Dominance as a 5th dimension. High-quality models that dominate many others rank highest. Low-quality models get penalized by being dominated. No magic thresholds — just principled math.

net_dom = Σ(my_Q × I_dominate) − Σ(their_Q × they_dominate_me)

Find the "elbow" where marginal quality gains require disproportionate cost increases. Maximum bang for your buck.

from typing import List, Optional

def binary_search(
    arr: List[int], 
    target: int
) -> Optional[int]:
    """Binary search implementation.
    
    Args:
        arr: Sorted list of integers
        target: Value to find
        
    Returns:
        Index if found, None otherwise
    """
    if not arr:
        raise ValueError("Empty array")
    
    left, right = 0, len(arr) - 1
    
    while left <= right:
        mid = (left + right) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    return None

Quick Start

Simple to Integrate

from llm_jury import get_value_recommendations

# Get optimal model for your constraints
recommendations = get_value_recommendations(
    prompt="Write a Python function to sort a list",
    use_case="coding",
    quality_threshold=0.85,  # 85% of baseline quality
    cost_threshold=0.30,     # 30% of baseline cost
)

# Get the top recommendation
best = recommendations[0]
print(f"Use {best.model_name}: {best.reasoning}")
from llm_jury import Optimizer, OptimizationStrategy, ModelRegistry

# Load models and set baseline
registry = ModelRegistry.load_cache()
baseline = registry.get_by_name("Gemini 3 Pro")

# Create optimizer with HYBRID strategy (default)
# Uses 5D Chebyshev with Net Pareto Dominance
optimizer = Optimizer(
    baseline_model=baseline,
    all_models_data=registry.to_list(),
    strategy=OptimizationStrategy.HYBRID,  # 5D with Net Dominance
)

# Rank models for a specific task
results = optimizer.rank(
    models=registry.all(),
    decision=RoutingDecision(category=PromptCategory.CODING),
    top_k=5,
)
from llm_jury import Optimizer, ObjectiveRegistry, Objective

# Create custom objective registry
registry = ObjectiveRegistry.default()

# Add a custom objective (e.g., ethics score)
registry.register(Objective(
    name="ethics",
    display_name="Ethics Score",
    direction="maximize",
    default_weight=0.15,
    extractor=lambda m, d, ctx: m.ethics_score,
))

# Use custom weights
optimizer = Optimizer(
    baseline_model=baseline,
    objectives=registry,
    custom_weights={
        "quality": 0.30,
        "cost": 0.25,
        "ethics": 0.20,
        "latency": 0.15,
        "hallucination": 0.10,
    }
)
