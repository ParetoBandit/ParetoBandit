\subsection{Algorithm Specification}
\label{appendix:algorithm}

This subsection provides pseudocode for the banditGPT routing system.
Algorithm~\ref{alg:routing} describes the top-level routing and feedback loop.
Algorithm~\ref{alg:corralling} details the Corralling meta-learner that
coordinates multiple expert strategies.
Algorithm~\ref{alg:linucb} specifies the Hybrid LinUCB expert used by both
the warmup and tabula rasa strategies.  Each expert decomposes the reward into
a family-shared component $\beta_F$ and an arm-specific residual $\theta_a$;
when every family is a singleton, the algorithm reduces to standard Disjoint
LinUCB.

% ============================================================================
% ALGORITHM 1: Top-level routing loop
% ============================================================================

\begin{algorithm}[h]
\caption{banditGPT Online Routing}
\label{alg:routing}
\begin{algorithmic}[1]

\REQUIRE Prompt stream $\{q_t\}_{t=1}^T$, model portfolio $\mathcal{M} = \{m_1, \ldots, m_K\}$
\REQUIRE PCA matrix $W \in \mathbb{R}^{32 \times 384}$ (trained offline on warmup data)
\REQUIRE Sentence encoder $\phi: \text{string} \to \mathbb{R}^{384}$
\REQUIRE Warmup expert $E_1$ (initialized from offline priors), Tabula rasa expert $E_2$ (cold start)
\REQUIRE Corralling meta-learner $\mathcal{C}$ with experts $(E_1, E_2)$

\medskip
\STATE \textbf{Initialize:} Expert weights $\mathbf{w}^{(0)} = [w_1^{(0)}, w_2^{(0)}]$ (e.g., uniform or prior-biased)

\medskip
\FOR{$t = 1, 2, \ldots, T$}
    \STATE \textbf{// Feature extraction}
    \STATE $e_t \leftarrow \phi(q_t)$ \hfill $\triangleright$ Encode prompt to 384-D embedding
    \STATE $z_t \leftarrow W \cdot e_t$ \hfill $\triangleright$ PCA compress to 32-D
    \STATE $x_t \leftarrow [z_t;\, 1] \in \mathbb{R}^{33}$ \hfill $\triangleright$ Append bias term
    
    \medskip
    \STATE \textbf{// Constraint filtering}
    \STATE $\mathcal{F}_t \leftarrow \{m \in \mathcal{M} : \text{cost}(m) \leq c_{\max},\; \text{latency}(m) \leq l_{\max},\; q_m \geq q_{\min}\}$
    
    \medskip
    \STATE \textbf{// Model selection via Corralling (Algorithm~\ref{alg:corralling})}
    \STATE $(a_t, i_t, p_{i_t}) \leftarrow \textsc{CorrallingSelect}(\mathcal{C}, x_t, \mathcal{F}_t, t)$
    \hfill $\triangleright$ $a_t$: model, $i_t$: expert index, $p_{i_t}$: selection probability
    
    \medskip
    \STATE \textbf{// Serve request and observe reward}
    \STATE $r_t \leftarrow \text{reward}(q_t, a_t)$ \hfill $\triangleright$ Binary win/loss from judge
    \STATE $r_t \leftarrow \text{clip}(r_t, 0, 1)$
    
    \medskip
    \STATE \textbf{// Two-level update (Algorithm~\ref{alg:corralling}, lines 12--20)}
    \STATE \textsc{CorrallingUpdate}$(\mathcal{C}, x_t, a_t, r_t, i_t, p_{i_t})$
\ENDFOR

\end{algorithmic}
\end{algorithm}

% ============================================================================
% ALGORITHM 2: Corralling meta-learner
% ============================================================================

\begin{algorithm}[h]
\caption{Corralling Meta-Learner (Log-Barrier OMD, Agarwal et al.\ 2017)}
\label{alg:corralling}
\begin{algorithmic}[1]

\REQUIRE $N$ experts $\{E_1, \ldots, E_N\}$, base learning rate $\eta_0$, mixing floor $\gamma$, loss decay $\delta$

\medskip
\STATE \textbf{Initialize:} $S_i \leftarrow 0$ for all $i \in [N]$ \hfill $\triangleright$ Cumulative squared losses
\STATE \textbf{Initialize:} $w_i \leftarrow 1/N$ for all $i \in [N]$ \hfill $\triangleright$ Normalized weights

\medskip
\FUNCTION{CorrallingSelect($\mathcal{C}, x_t, \mathcal{F}_t, t$)}
    \STATE $p_i^{(t)} \leftarrow (1 - \gamma) \cdot w_i + \gamma / N$ for all $i$ \hfill $\triangleright$ Mix with uniform (Eq.~\ref{eq:corralling_update})
    \STATE Sample expert $i_t \sim \text{Categorical}(p_1^{(t)}, \ldots, p_N^{(t)})$
    \STATE $a_t \leftarrow E_{i_t}.\text{SelectArm}(x_t, \mathcal{F}_t, t)$ \hfill $\triangleright$ Ask chosen expert (Alg.~\ref{alg:linucb})
    \RETURN $(a_t, i_t, p_{i_t}^{(t)})$
\ENDFUNCTION

\medskip
\FUNCTION{CorrallingUpdate($\mathcal{C}, x_t, a_t, r_t, i_t, p_{i_t}$)}
    \STATE \textbf{// Level 1: Meta-weight update (Log-Barrier OMD)}
    \STATE $\ell_t \leftarrow 1 - r_t$ \hfill $\triangleright$ Convert reward to loss
    \STATE $\hat{\ell}_{i_t} \leftarrow \ell_t / p_{i_t}$ \hfill $\triangleright$ Importance-weighted estimator (bounded by $N/\gamma$)
    \STATE $S_i \leftarrow \delta^2 \cdot S_i$ for all $i \in [N]$ \hfill $\triangleright$ Decay squared losses ($\delta = 1$ in experiments)
    \STATE $S_{i_t} \leftarrow S_{i_t} + \hat{\ell}_{i_t}^2$ \hfill $\triangleright$ Accumulate squared loss for endorsing expert
    \STATE $\eta_i \leftarrow \eta_0 \,/\, \sqrt{S_i + \epsilon}$ for all $i$ \hfill $\triangleright$ Per-expert adaptive learning rate
    \STATE $w_i \leftarrow w_i \,/\, (1 + \eta_i \cdot \hat{\ell}_i \cdot w_i)$ for all $i$ \hfill $\triangleright$ Log-Barrier OMD update
    \STATE $w_i \leftarrow w_i \,/\, \sum_j w_j$ \hfill $\triangleright$ Project onto simplex
    
    \medskip
    \STATE \textbf{// Level 2: All experts learn from observation}
    \FOR{$j = 1, \ldots, N$}
        \STATE $E_j.\text{Update}(x_t, a_t, r_t)$ \hfill $\triangleright$ LinUCB update (Alg.~\ref{alg:linucb})
    \ENDFOR
\ENDFUNCTION

\end{algorithmic}
\end{algorithm}

% ============================================================================
% ALGORITHM 3: Hybrid LinUCB expert (family-shared + arm-specific)
% ============================================================================

\begin{algorithm}[h]
\caption{Cost-Aware Hybrid LinUCB Expert}
\label{alg:linucb}
\begin{algorithmic}[1]

\REQUIRE Arms (models) $\mathcal{M}$, family mapping $F: \mathcal{M} \to \mathcal{G}$, dimension $d$, \\
\hspace{2.5em} exploration coefficient schedule $\alpha_t$, cost penalty $\lambda_c$, regularization $\lambda$, \\
\hspace{2.5em} optional warmup priors $\{(\theta_a^{(0)},\, n_{\text{eff},a})\}_{a \in \mathcal{M}}$

\medskip
\STATE \textbf{Initialize} family-shared state for each family $g \in \mathcal{G}$:
\STATE $A_g^{(0)} \leftarrow \lambda \cdot I_d$;\quad $b_g^{(0)} \leftarrow \mathbf{0}$;\quad $(A_g^{(0)})^{-1} \leftarrow (\lambda \cdot I_d)^{-1}$

\medskip
\STATE \textbf{Initialize} arm-specific state for each arm $a \in \mathcal{M}$:
\IF{warmup priors available}
    \STATE $A_a \leftarrow n_{\text{eff},a} \cdot \lambda \cdot I_d$;\quad $b_a \leftarrow n_{\text{eff},a} \cdot \lambda \cdot \theta_a^{(0)}$ \hfill $\triangleright$ Bayesian prior injection
\ELSE
    \STATE $A_a \leftarrow \lambda \cdot I_d$;\quad $b_a \leftarrow \mathbf{0}$ \hfill $\triangleright$ Tabula rasa
\ENDIF
\STATE $A_a^{-1} \leftarrow A_a^{-1}$ \hfill $\triangleright$ Cache inverse for $O(d^2)$ updates

\medskip
\FUNCTION{SelectArm($x_t, \mathcal{F}_t, t$)}
    \STATE $\alpha \leftarrow \alpha_{\text{start}} + (\alpha_{\text{end}} - \alpha_{\text{start}}) \cdot t / T$ \hfill $\triangleright$ Linear decay schedule
    \FOR{each $a \in \mathcal{F}_t$}
        \STATE $g \leftarrow F(a)$ \hfill $\triangleright$ Look up family for arm $a$
        \STATE $\hat{\beta}_g \leftarrow (A_g^{(0)})^{-1} b_g^{(0)}$ \hfill $\triangleright$ Family-shared estimate
        \STATE $\hat{\theta}_a \leftarrow A_a^{-1} b_a$ \hfill $\triangleright$ Arm-specific residual estimate
        \STATE $\hat{r}_a \leftarrow x_t^\top \hat{\beta}_g + x_t^\top \hat{\theta}_a$ \hfill $\triangleright$ Hybrid predicted reward
        \STATE $\sigma_a^2 \leftarrow \underbrace{x_t^\top (A_g^{(0)})^{-1} x_t}_{\text{shared variance}} + \underbrace{x_t^\top A_a^{-1} x_t}_{\text{arm variance}}$ \hfill $\triangleright$ Combined uncertainty
        \STATE $\text{UCB}_a \leftarrow \hat{r}_a + \alpha \cdot \sqrt{\sigma_a^2} - \lambda_c \cdot \tilde{c}_a$ \hfill $\triangleright$ Cost-penalized UCB score
    \ENDFOR
    \RETURN $\arg\max_{a \in \mathcal{F}_t} \text{UCB}_a$ \hfill $\triangleright$ Random tie-breaking
\ENDFUNCTION

\medskip
\FUNCTION{Update($x_t, a_t, r_t$)}
    \STATE $g \leftarrow F(a_t)$ \hfill $\triangleright$ Family of chosen arm
    \STATE \textbf{// Step 1: Compute residual from current shared estimate}
    \STATE $\hat{\beta}_g \leftarrow (A_g^{(0)})^{-1} b_g^{(0)}$
    \STATE $\tilde{r}_t \leftarrow r_t - x_t^\top \hat{\beta}_g$ \hfill $\triangleright$ Residual reward
    
    \medskip
    \STATE \textbf{// Step 2: Update family-shared matrices (full reward)}
    \STATE $v_g \leftarrow (A_g^{(0)})^{-1} x_t$
    \STATE $(A_g^{(0)})^{-1} \leftarrow (A_g^{(0)})^{-1} - \frac{v_g \, v_g^\top}{1 + x_t^\top v_g}$ \hfill $\triangleright$ Sherman--Morrison
    \STATE $A_g^{(0)} \leftarrow A_g^{(0)} + x_t \, x_t^\top$
    \STATE $b_g^{(0)} \leftarrow b_g^{(0)} + r_t \cdot x_t$
    
    \medskip
    \STATE \textbf{// Step 3: Update arm-specific matrices (residual reward)}
    \STATE $v_a \leftarrow A_{a_t}^{-1} x_t$
    \STATE $A_{a_t}^{-1} \leftarrow A_{a_t}^{-1} - \frac{v_a \, v_a^\top}{1 + x_t^\top v_a}$ \hfill $\triangleright$ Sherman--Morrison
    \STATE $A_{a_t} \leftarrow A_{a_t} + x_t \, x_t^\top$
    \STATE $b_{a_t} \leftarrow b_{a_t} + \tilde{r}_t \cdot x_t$ \hfill $\triangleright$ Learn residual, not raw reward
\ENDFUNCTION

\end{algorithmic}
\end{algorithm}

\paragraph{Notation summary.}
$x_t \in \mathbb{R}^{33}$ is the PCA-compressed context vector (32 components + bias);
$a_t \in \mathcal{M}$ is the selected model;
$r_t \in [0, 1]$ is the observed reward;
$i_t \in [N]$ is the expert chosen by the meta-learner;
$p_{i_t}$ is the probability with which expert $i_t$ was selected;
$\hat{\ell}$ denotes the importance-weighted loss estimator;
$\tilde{c}_a$ is the normalized cost for model $a$;
$F: \mathcal{M} \to \mathcal{G}$ maps each arm to its family (inferred from model
provider, e.g., GPT-4-Turbo and GPT-4o share family ``openai/gpt-4'');
$\hat{\beta}_g$ is the family-shared reward parameter for family $g$;
$\hat{\theta}_a$ is the arm-specific residual parameter;
$A_g^{(0)}, b_g^{(0)}$ are the family-shared precision matrix and moment vector;
$\tilde{r}_t = r_t - x_t^\top \hat{\beta}_g$ is the residual reward used for
arm-specific updates.

\paragraph{Key design decisions.}
\begin{enumerate}[leftmargin=*,noitemsep]
    \item \textbf{All experts learn} (Algorithm~\ref{alg:corralling}, lines~18--20):
    Every expert observes every $(x_t, a_t, r_t)$ tuple, regardless of which expert
    was selected.  This prevents data starvation of the non-selected expert and
    ensures the tabula rasa expert can serve as a viable fallback under prior mismatch.
    
    \item \textbf{Constraint filtering before selection} (Algorithm~\ref{alg:routing},
    line~9): Hard constraints (cost, latency) are enforced before the bandit scores
    arms, guaranteeing per-request constraint satisfaction.
    
    \item \textbf{$\gamma$-mixing prevents expert death}
    (Algorithm~\ref{alg:corralling}, line~4): Each expert retains probability
    $\geq \gamma/N$ at every round, ensuring bounded recovery time after
    distribution shifts (Theorem~\ref{thm:recovery}).
    
    \item \textbf{Reward clamping} (Algorithm~\ref{alg:routing}, line~15):
    Rewards are clipped to $[0, 1]$ before the importance-weighted loss computation,
    bounding $\hat{\ell}_{i_t} \leq N/\gamma$
    (Corollary~\ref{cor:bounded_iw}).
    
    \item \textbf{Hybrid family decomposition}
    (Algorithm~\ref{alg:linucb}): The reward model
    $\mathbb{E}[r \mid x, a] = x^\top \beta_{F(a)} + x^\top \theta_a$
    separates a family-shared component $\beta_g$ (updated from \emph{all}
    observations within the family) from an arm-specific residual $\theta_a$.
    When a new model joins an existing family, the shared $\beta_g$ is
    already trained, providing immediate transfer.  For singleton families
    the residual converges to near-zero and the policy degenerates gracefully
    to family-only (effectively Disjoint) behaviour.
\end{enumerate}
