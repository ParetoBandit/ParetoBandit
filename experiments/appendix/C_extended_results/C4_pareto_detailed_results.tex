\paragraph{Key metrics.}
\begin{itemize}[leftmargin=*,noitemsep]
    \item \textbf{banditGPT peak quality:} 0.9153 at cost \$0.00985
          ($\lambda{=}0$, 20 seeds, 95\% CI $\pm$0.0027).
    \item \textbf{RouteLLM-MF peak quality:} 0.8827 at cost \$0.00651
          (deterministic).
    \item \textbf{Oracle:} 0.9533 at cost \$0.00195 (perfect per-prompt routing).
    \item \textbf{Gap closure:} banditGPT closes
          $\frac{0.9153 - 0.8227}{0.9533 - 0.8227} = 70.9\%$ of the
          Oracle--Mixtral gap, vs.\ RouteLLM-MF's
          $\frac{0.8827 - 0.8227}{0.9533 - 0.8227} = 45.9\%$.
\end{itemize}
